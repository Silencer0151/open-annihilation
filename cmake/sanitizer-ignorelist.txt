# Where an OA_SANITIZERS build with Clang leaves alignment unchecked
# (-fsanitize-ignorelist). The canonical records keep their packed layouts,
# so whether a field sits at an address its type aligns to depends on where
# its record sits. The code below still binds references to fields that the
# World's tables place at aligned addresses, and would be reported for a
# record placed elsewhere. Everywhere else, the readers of game files
# included, a misaligned load or reference stops the test that makes it.
# Drop a directory once its code copies packed fields out instead of reading
# them in place.
#
# The standard library headers are listed too: a misaligned reference handed
# to them is reported where the caller binds it, unless the caller is listed.
#
# Compilers read each path either as a glob or as a regular expression, so
# a path names a literal '+' as the class [+], which both readers take as
# that one character.
[alignment]
src:*/src/app/*
src:*/src/sim/match-runtime/*
src:*/src/sim/unit-spawn/*
src:*/include/c[+][+]/*
