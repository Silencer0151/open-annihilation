cmake_minimum_required(VERSION 3.24)
project(oa_sim_unit_health LANGUAGES CXX)
include(CTest)
add_library(oa-sim-unit-health src/unit_health.cpp src/paralysis.cpp src/veterancy.cpp)
add_library(oa::sim::unit_health ALIAS oa-sim-unit-health)
target_link_libraries(oa-sim-unit-health PRIVATE oa-base-game-math)
target_include_directories(oa-sim-unit-health PUBLIC include)
target_compile_features(oa-sim-unit-health PUBLIC cxx_std_20)
if(TARGET oa-options)
  target_link_libraries(oa-sim-unit-health PUBLIC oa-options)
endif()
target_link_libraries(oa-sim-unit-health PUBLIC oa-core-types oa-data-match-rules)
if(BUILD_TESTING)
  add_executable(oa-unit-health-test tests/unit_health_test.cpp)
  target_link_libraries(oa-unit-health-test PRIVATE oa-sim-unit-health)
  add_test(NAME unit-health COMMAND oa-unit-health-test)
  add_executable(oa-unit-paralysis-test tests/paralysis_test.cpp)
  target_link_libraries(oa-unit-paralysis-test PRIVATE oa-sim-unit-health)
  add_test(NAME unit-paralysis COMMAND oa-unit-paralysis-test)
  add_executable(oa-unit-health-veterancy-test tests/veterancy_test.cpp)
  target_link_libraries(oa-unit-health-veterancy-test PRIVATE oa-sim-unit-health oa-test-support)
  add_test(NAME unit-health-veterancy COMMAND oa-unit-health-veterancy-test)
  add_executable(oa-unit-health-repair-rate-test tests/repair_rate_test.cpp)
  target_link_libraries(oa-unit-health-repair-rate-test PRIVATE oa-sim-unit-health oa-test-support)
  add_test(NAME unit-health-repair-rate COMMAND oa-unit-health-repair-rate-test)
endif()
