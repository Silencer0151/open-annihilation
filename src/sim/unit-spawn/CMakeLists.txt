cmake_minimum_required(VERSION 3.24)
project(oa_sim_unit_spawn LANGUAGES CXX)
include(CTest)
add_library(oa-sim-unit-spawn src/spawn.cpp src/legacy_views.cpp)
add_library(oa::sim::unit_spawn ALIAS oa-sim-unit-spawn)
target_include_directories(oa-sim-unit-spawn PUBLIC include)
target_link_libraries(oa-sim-unit-spawn PUBLIC oa-sim-spatial-state)
target_link_libraries(oa-sim-unit-spawn PUBLIC oa-sim-simulation-state)
target_compile_features(oa-sim-unit-spawn PUBLIC cxx_std_20)
if(TARGET oa-options)
  target_link_libraries(oa-sim-unit-spawn PUBLIC oa-options)
endif()
if(BUILD_TESTING)
  add_executable(oa-unit-spawn-test tests/spawn_test.cpp)
  target_link_libraries(oa-unit-spawn-test PRIVATE oa-sim-unit-spawn)
  add_test(NAME unit-spawn COMMAND oa-unit-spawn-test)
endif()
add_library(oa-sim-unit-spawn-runtime src/runtime.cpp)
add_library(oa::sim::unit_spawn_runtime ALIAS oa-sim-unit-spawn-runtime)
target_include_directories(oa-sim-unit-spawn-runtime PUBLIC include)
target_link_libraries(oa-sim-unit-spawn-runtime PUBLIC oa-sim-unit-spawn oa-data-unit-definitions oa-formats-objects3d oa-formats-cob
  PRIVATE oa-data-defs)
target_compile_features(oa-sim-unit-spawn-runtime PUBLIC cxx_std_20)
if(TARGET oa-options)
  target_link_libraries(oa-sim-unit-spawn-runtime PUBLIC oa-options)
endif()
if(BUILD_TESTING)
  add_executable(oa-unit-runtime-test tests/runtime_test.cpp)
  target_link_libraries(oa-unit-runtime-test PRIVATE oa-sim-unit-spawn-runtime)
  add_test(NAME unit-runtime COMMAND oa-unit-runtime-test)
endif()
